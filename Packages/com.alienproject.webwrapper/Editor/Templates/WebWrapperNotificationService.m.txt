#import <Foundation/Foundation.h>
#import <UserNotifications/UserNotifications.h>

@interface WebWrapperNotificationService : UNNotificationServiceExtension
@property(nonatomic, copy) void (^contentHandler)(UNNotificationContent *content);
@property(nonatomic, strong) UNMutableNotificationContent *bestAttemptContent;
@property(nonatomic, strong) NSURLSession *downloadSession;
@end

@implementation WebWrapperNotificationService

- (void)didReceiveNotificationRequest:(UNNotificationRequest *)request
                  withContentHandler:(void (^)(UNNotificationContent *content))contentHandler {
    self.contentHandler = contentHandler;
    self.bestAttemptContent = [request.content mutableCopy];
    NSDictionary *payload = request.content.userInfo;
    id options = payload[@"fcm_options"];
    id imageValue = [options isKindOfClass:NSDictionary.class] ? options[@"image"] : nil;
    if (![imageValue isKindOfClass:NSString.class]) {
        imageValue = payload[@"image"] ?: payload[@"gcm.notification.image"];
    }
    NSURL *imageURL = [imageValue isKindOfClass:NSString.class] ? [NSURL URLWithString:imageValue] : nil;
    if (![imageURL.scheme.lowercaseString isEqualToString:@"https"] || imageURL.host.length == 0) {
        [self finish];
        return;
    }

    NSURLSessionConfiguration *configuration = NSURLSessionConfiguration.ephemeralSessionConfiguration;
    configuration.timeoutIntervalForRequest = 15;
    configuration.timeoutIntervalForResource = 20;
    configuration.requestCachePolicy = NSURLRequestReloadIgnoringLocalCacheData;
    self.downloadSession = [NSURLSession sessionWithConfiguration:configuration];
    NSURLSessionDownloadTask *task = [self.downloadSession downloadTaskWithURL:imageURL
        completionHandler:^(NSURL *location, NSURLResponse *response, NSError *error) {
            @synchronized(self) {
                if (!self.contentHandler) { return; }
                NSInteger status = [response isKindOfClass:NSHTTPURLResponse.class]
                    ? ((NSHTTPURLResponse *)response).statusCode : 0;
                if (!error && location && status >= 200 && status < 300) {
                    [self attachImageAtURL:location response:response originalURL:imageURL];
                }
            }
            [self finish];
        }];
    [task resume];
}

- (void)attachImageAtURL:(NSURL *)location response:(NSURLResponse *)response originalURL:(NSURL *)originalURL {
    NSNumber *size = nil;
    [location getResourceValue:&size forKey:NSURLFileSizeKey error:nil];
    if (!size || size.unsignedLongLongValue == 0 || size.unsignedLongLongValue > 10 * 1024 * 1024) {
        return;
    }
    NSDictionary *extensionsByMimeType = @{
        @"image/jpeg": @"jpg", @"image/png": @"png", @"image/gif": @"gif"
    };
    NSString *extension = extensionsByMimeType[response.MIMEType.lowercaseString];
    if (!extension) {
        NSString *candidate = originalURL.pathExtension.lowercaseString;
        if ([@[@"jpg", @"jpeg", @"png", @"gif"] containsObject:candidate]) { extension = candidate; }
    }
    if (!extension) { return; }

    NSURL *directory = [NSURL fileURLWithPath:[NSTemporaryDirectory()
        stringByAppendingPathComponent:NSUUID.UUID.UUIDString] isDirectory:YES];
    NSFileManager *manager = NSFileManager.defaultManager;
    if (![manager createDirectoryAtURL:directory withIntermediateDirectories:YES attributes:nil error:nil]) { return; }
    NSURL *destination = [directory URLByAppendingPathComponent:[@"image" stringByAppendingPathExtension:extension]];
    if (![manager moveItemAtURL:location toURL:destination error:nil]) {
        [manager removeItemAtURL:directory error:nil];
        return;
    }
    UNNotificationAttachment *attachment = [UNNotificationAttachment attachmentWithIdentifier:@"image"
        URL:destination options:nil error:nil];
    if (attachment) {
        self.bestAttemptContent.attachments = [self.bestAttemptContent.attachments arrayByAddingObject:attachment];
    } else {
        [manager removeItemAtURL:directory error:nil];
    }
}

- (void)finish {
    void (^handler)(UNNotificationContent *) = nil;
    UNNotificationContent *content = nil;
    @synchronized(self) {
        handler = self.contentHandler;
        self.contentHandler = nil;
        content = self.bestAttemptContent;
    }
    if (handler) { handler(content); }
    [self.downloadSession invalidateAndCancel];
}

- (void)serviceExtensionTimeWillExpire {
    [self finish];
}

@end
